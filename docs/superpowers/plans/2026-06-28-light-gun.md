# Light Gun Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a first-person light-placement tool to the FPS walk mode: shoot or drop point lights that expire over time, wired into the existing clustered GPU lighting pipeline.

**Architecture:** Reserve 33 tail slots of the 256-slot clustered-lights buffer (1 projectile + 32 placed) and expose `setLightDirect`/`clearLight` to write them from JS each frame. Light gun state (lgParams, placedLights[], slotFree[]) lives alongside the existing FPS walk state in environment-viewer.html. A per-frame `updatePlacedLights(dt,t)` updates physics/drift/fade/expiry and flushes to GPU before the cull pass.

**Tech Stack:** Three.js WebGPU, TSL nodes, vanilla JS/DOM, existing `fpHeader`/`fpSlider` FPS panel helpers.

function lgRaycastTerrain() {
  const dir = new THREE.Vector3();
  camera.getWorldDirection(dir);
  const ox = camera.position.x, oy = camera.position.y, oz = camera.position.z;
  for (let i = 1; i <= LG_RAY_STEPS; i++) {
    const t = (i / LG_RAY_STEPS) * LG_RAY_MAX_DIST;
    const x = ox + dir.x * t, y = oy + dir.y * t, z = oz + dir.z * t;
    if (y <= terrainHeight(x, z)) return { x, y: terrainHeight(x, z), z, hit: true };
  }
  return { hit: false };
}

function lgUpdateLights(dt, t) {
  if (!clusteredLightsRef) return;

  // In-flight projectile
  if (lgInFlight) {
    if (lgInFlight.arc) lgInFlight.vy -= LG_FALL_GRAVITY * dt;
    lgInFlight.x += lgInFlight.vx * dt;
    lgInFlight.y += lgInFlight.vy * dt;
    lgInFlight.z += lgInFlight.vz * dt;
    const outOfBounds = Math.abs(lgInFlight.x) > 260 || Math.abs(lgInFlight.z) > 260;
    const th = terrainHeight(lgInFlight.x, lgInFlight.z);
    if (lgInFlight.y <= th || outOfBounds) {
      clusteredLightsRef.clearLight(LG_PROJECTILE_SLOT);
      if (!outOfBounds) {
        const slot = lgFindFreeSlot();
        if (slot !== -1) {
          const yPlace = th + (lgInFlight.float ? 1.5 : 0.2);
          placedLights.push({
            slot, freeIdx: slot - LG_FIRST_SLOT,
            x: lgInFlight.x, y: yPlace, z: lgInFlight.z, vy: 0,
            grounded: lgInFlight.float,
            lifespan: lgInFlight.lifespan, totalLife: lgInFlight.totalLife,
            r: lgInFlight.r, g: lgInFlight.g, b: lgInFlight.b,
            brightness: lgInFlight.brightness, radius: lgInFlight.radius,
            float: lgInFlight.float, drift: lgInFlight.drift,
            driftPhase: lgInFlight.driftPhase,
          });
          lgSlotFree[slot - LG_FIRST_SLOT] = false;
        }
      }
      lgInFlight = null;
    } else {
      clusteredLightsRef.setLightDirect(LG_PROJECTILE_SLOT, {
        x: lgInFlight.x, y: lgInFlight.y, z: lgInFlight.z, radius: lgInFlight.radius,
        r: lgInFlight.r, g: lgInFlight.g, b: lgInFlight.b, intensity: lgInFlight.brightness,
      });
    }
  }

  // Placed lights
  for (let i = placedLights.length - 1; i >= 0; i--) {
    const L = placedLights[i];
    if (!L.float && !L.grounded) {
      L.vy -= LG_FALL_GRAVITY * dt;
      L.y += L.vy * dt;
      const th = terrainHeight(L.x, L.z);
      if (L.y <= th) { L.y = th; L.vy = 0; L.grounded = true; }
    }
    let xOff = 0, zOff = 0;
    if (L.float && L.drift) {
      xOff = Math.sin(t * 0.3 + L.driftPhase) * 3;
      zOff = Math.cos(t * 0.27 + L.driftPhase) * 3;
    }
    L.lifespan -= dt;
    const fadeScale = L.lifespan < LG_FADE_WINDOW ? Math.max(0, L.lifespan / LG_FADE_WINDOW) : 1;
    if (L.lifespan <= 0) {
      clusteredLightsRef.clearLight(L.slot);
      lgSlotFree[L.freeIdx] = true;
      placedLights.splice(i, 1);
      continue;
    }
    clusteredLightsRef.setLightDirect(L.slot, {
      x: L.x + xOff, y: L.y, z: L.z + zOff, radius: L.radius,
      r: L.r, g: L.g, b: L.b, intensity: L.brightness * fadeScale,
    });
  }
}
